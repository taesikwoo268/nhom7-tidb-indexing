\subsection{Khuyến nghị thiết kế chỉ mục}

\begin{enumerate}
  \item \textbf{Khóa chính số nguyên clustered phù hợp cho bảng vận đơn khi không có yêu cầu đặt mã vận đơn làm key.} RQ3 cho thấy mã vận đơn làm \code{idx\_sender\_created} tăng 27,4\% trên MySQL và 39,5\% trên TiDB so với khóa \code{id}; trên TiDB, tra theo key không clustered cũng cần thêm một lần gọi TiKV.

  \item \textbf{Chỉ tạo covering index cho truy vấn mang lại lợi ích đọc đủ lớn.} Trên M2, covering Q2 ở $R=20$ nhanh hơn 2,24 lần trên TiDB và 1,12 lần trên MySQL. Khi W2 cập nhật \code{status}, \code{idx\_sender\_cov} làm TPS TiDB C3 giảm 61,6\% so với \code{idx\_sender\_created}; MySQL giảm 29,5\%. Không nên thêm cột chỉ để dự kiến vài truy vấn tăng tốc nếu cột đó nằm trong tải ghi định kỳ.

  \item \textbf{Không dùng một ngưỡng chung cho hai engine.} Với truy vấn \code{cod\_amount}, MySQL nên kiểm tra lực cản của chỉ mục quanh mức 5--10\%; TiDB nên quan sát lại ở mức 2--5\%. Trong triển khai, số liệu thực tế và thống kê optimizer của từng bảng cần được kiểm tra, không nên chỉ dùng số crossover của thí nghiệm này làm tham số cố định.

  \item \textbf{Ưu tiên MySQL InnoDB cho quy mô bảng nằm trong tài nguyên một máy, truy vấn nhỏ và áp lực ghi thấp đến trung bình.} Trong RQ4, MySQL duy trì TPS cao hơn TiDB C3 cả ở tải 0 và 5 chỉ mục. Cấu hình này cũng có độ trễ p95 13 ms ở bộ 5, thấp hơn 204 ms của TiDB C3 cho W1.

  \item \textbf{Ưu tiên TiDB khi giá trị thiết kế là mở rộng dữ liệu và dịch vụ phân tán, không chỉ TPS trên một máy.} TiDB cung cấp SQL quan hệ, chia Region tự động và nhân bản C3; đánh đổi là chi phí giao dịch, nhân bản và chỉ mục KV làm hiệu năng ghi giảm mạnh trong hồ sơ XS. C1 chỉ dùng để tách chi phí nhân bản: C1 vẫn giảm 79,4\% ở W1 và 51,6\% ở W2 từ bộ 0 đến bộ 5.

  \item \textbf{Tránh đánh chỉ mục phụ quá sớm cho workload ghi W1.} Khi thêm 5 chỉ mục, TPS W1 giảm 82,7\% trên TiDB C3 và 18,9\% trên MySQL. Với mỗi chỉ mục, nhóm cần xác định truy vấn đọc bù được chi phí ghi, dung lượng và thời gian tối thiểu để duy trì/sửa chỉ mục.
\end{enumerate}
