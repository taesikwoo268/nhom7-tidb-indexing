\subsection{Khuyến nghị thiết kế chỉ mục}

\begin{enumerate}
  \item \textbf{Dùng khóa chính số nguyên clustered, trừ khi phải tra theo mã vận đơn nhiều nhất.} RQ3 cho thấy dùng mã vận đơn làm khóa chính làm \code{idx\_sender\_created} lớn thêm 27,4\% trên MySQL và 39,5\% trên TiDB so với khóa \code{id}. Trên TiDB, tra theo một khóa không clustered, kể cả khóa chính nonclustered, cần thêm một lần gọi TiKV.

  \item \textbf{Chỉ làm covering index khi lợi ích đọc đủ lớn và cột thêm vào ít bị cập nhật.} Trên M2, covering làm Q2 ($R = 20$) nhanh gấp 2,24 lần trên TiDB nhưng chỉ 1,12 lần trên MySQL. Đổi lại, chỉ mục lớn thêm khoảng 22\% trên MySQL và 36\% trên TiDB, và khi \code{status} bị cập nhật, TPS giảm 29,5\% trên MySQL và 61,6\% trên TiDB C3. Ngưỡng chuyển sang quét bảng cũng khác nhau giữa hai hệ (khoảng 6,9\% trên MySQL, 3,2\% trên TiDB), nên cần kiểm tra lại trên dữ liệu thật thay vì dùng một con số chung.

  \item \textbf{Cân nhắc từng chỉ mục phụ trên TiDB vì chi phí ghi lớn.} Chỉ riêng chỉ mục đầu tiên đã làm TPS thêm đơn của TiDB C3 giảm 50,3\%; với 5 chỉ mục, mức giảm là 82,7\% so với 18,9\% trên MySQL. Mỗi chỉ mục chỉ nên được thêm khi có truy vấn đọc thực sự bù được chi phí ghi và dung lượng của nó.
\end{enumerate}
